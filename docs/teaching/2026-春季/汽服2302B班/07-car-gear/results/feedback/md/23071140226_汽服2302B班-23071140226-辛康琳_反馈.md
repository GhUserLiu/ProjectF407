**代码示例**:
```c
// 档位状态机实现

// 1. 定义状态枚举（顺序很重要）
typedef enum {
    GEAR_P = 0,  // 驻车档
    GEAR_R,       // 倒档
    GEAR_N,       // 空档
    GEAR_D        // 前进档
} GearState_t;

// 2. 当前状态
GearState_t current_gear = GEAR_P;

// 3. 状态切换函数（按键触发）
void Gear_Switch(void) {
    // 方法1: 循环切换
    current_gear = (current_gear + 1) % 4;
    Gear_Update_LED();
}

// 4. LED显示更新
void Gear_Update_LED(void) {
    switch(current_gear) {
        case GEAR_P:  // P: LED0亮，LED1灭
            HAL_GPIO_WritePin(LED0_GPIO_Port, LED0_Pin, GPIO_PIN_RESET);
            HAL_GPIO_WritePin(LED1_GPIO_Port, LED1_Pin, GPIO_PIN_SET);
            break;
        case GEAR_R:  // R: LED0灭，LED1亮
            HAL_GPIO_WritePin(LED0_GPIO_Port, LED0_Pin, GPIO_PIN_SET);
            HAL_GPIO_WritePin(LED1_GPIO_Port, LED1_Pin, GPIO_PIN_RESET);
            break;
        case GEAR_N:  // N: 都灭
            HAL_GPIO_WritePin(LED0_GPIO_Port, LED0_Pin, GPIO_PIN_SET);
            HAL_GPIO_WritePin(LED1_GPIO_Port, LED1_Pin, GPIO_PIN_SET);
            break;
        case GEAR_D:  // D: 都亮
            HAL_GPIO_WritePin(LED0_GPIO_Port, LED0_Pin, GPIO_PIN_RESET);
            HAL_GPIO_WritePin(LED1_GPIO_Port, LED1_Pin, GPIO_PIN_RESET);
            break;
    }
}

// 思考题：D档如何直接切到R档？
// 答案：实现两段切换
// D档按一次→N档，再按一次→R档
```

### DWT消抖原理未说明
**改进步骤**:
1. 说明DWT计数器频率为168MHz（系统频率）
1. 计算公式：消抖周期 = 系统频率 × 消抖时间
1. 50ms消抖需要：168MHz × 0.05s = 8,400,000周期
1. 对比软件延时消抖会阻塞CPU的问题

**代码示例**:
```c
// DWT消抖原理与实现

// 1. DWT初始化（使能周期计数器）
void DWT_Init(void) {
    CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;  // 使能DWT
    DWT->CYCCNT = 0;                                  // 复位计数器
    DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;             // 使能计数
}

// 2. 定义消抖周期数（50ms @ 168MHz）
// 计算公式：168,000,000 Hz × 0.05 s = 8,400,000
#define DEBOUNCE_CYCLES 8400000

// 3. 消抖检查函数
static uint32_t last_tick = 0;

int is_debounced(void) {
    uint32_t current = DWT->CYCCNT;
    // 检查是否超过消抖周期
    if ((current - last_tick) >= DEBOUNCE_CYCLES) {
        last_tick = current;
        return 1;  // 可以处理
    }
    return 0;  // 仍在消抖中
}

// DWT消抖优势：
// 1. 不阻塞CPU（不使用延时函数）
// 2. 精度高（基于硬件计数器）
// 3. 开销小（仅读取寄存器）
```
